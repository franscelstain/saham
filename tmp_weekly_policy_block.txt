        // --- policy rules ---

        $tickPct = ($close > 0) ? ($tickSize / $close) : null;
        $atrPct = ($atr14 !== null && $close > 0) ? ((float)$atr14 / $close) : null;

        $eligBlockCodes = [];
        $entryStyle = 'Default';
        $scoreTotal = 0.0;
        $derived = [
            'dv20_idr' => ($dv20 !== null && is_numeric($dv20)) ? (int) round((float)$dv20) : null,
            'atr_pct'  => ($atrPct !== null) ? round((float)$atrPct, 4) : null,
            'tick_pct' => ($tickPct !== null) ? round((float)$tickPct, 6) : null,
        ];

        if ($policy === 'WEEKLY_SWING') {
            $hh20 = (isset($r['hh20']) && is_numeric($r['hh20'])) ? (float)$r['hh20'] : null;
            $ll5  = (isset($r['ll5']) && is_numeric($r['ll5'])) ? (float)$r['ll5'] : null;
            $roc20 = (isset($r['roc20']) && is_numeric($r['roc20'])) ? (float)$r['roc20'] : null;

            // resistance_20 = highest_high(20) + tick
            $resistance20 = ($hh20 !== null) ? ($hh20 + $tickSize) : null;

            if ($resistance20 === null) {
                $reasonCodes[] = 'WS_MISSING_HH20';
                $eligBlockCodes[] = 'WS_MISSING_HH20';
            } else {
                // setup_type ∈ {BREAKOUT, PULLBACK}
                $setupType = ($close >= ($resistance20 - $tickSize)) ? 'BREAKOUT' : 'PULLBACK';
                $entryStyle = ($setupType === 'BREAKOUT') ? 'Breakout' : 'Pullback';
            }

            // 2.1 Trend / Structure gate
            if ($ma20 === null || $ma50 === null || (float)$ma20 <= 0 || (float)$ma50 <= 0 || $resistance20 === null) {
                $reasonCodes[] = 'WS_TREND_MISSING_MA';
                $eligBlockCodes[] = 'WS_TREND_MISSING_MA';
            } else {
                $ma20f = (float)$ma20;
                $ma50f = (float)$ma50;
                $condA = ($close >= $ma20f) && ($ma20f >= $ma50f);
                $condB = ($close >= $resistance20) && ($ma20f >= $ma50f);
                if (!$condA && !$condB) {
                    $reasonCodes[] = 'WS_TREND_NOT_OK';
                    $eligBlockCodes[] = 'WS_TREND_NOT_OK';
                }
            }

            // 2.2 Liquidity / Vol / Tick gates
            if ($dv20 === null || (float)$dv20 < 5000000000.0) {
                $reasonCodes[] = 'WS_MIN_DV20_FAIL';
                $eligBlockCodes[] = 'WS_MIN_DV20_FAIL';
            }
            if ($atrPct === null || (float)$atrPct < 0.02 || (float)$atrPct > 0.20) {
                $reasonCodes[] = 'WS_ATR_PCT_OUT_OF_RANGE';
                $eligBlockCodes[] = 'WS_ATR_PCT_OUT_OF_RANGE';
            }
            if ($tickPct === null || (float)$tickPct > 0.015) {
                $reasonCodes[] = 'WS_TICK_PCT_TOO_WIDE';
                $eligBlockCodes[] = 'WS_TICK_PCT_TOO_WIDE';
            }

            // Build plan levels (2.3)
            $levels = $this->buildLevelsWeekly($setupType, $r, $tickSize, $resistance20, $ll5);

            // 2.4 Stop validity & RR gate
            $entry = $levels['entry_trigger_price'] ?? null;
            $sl    = $levels['stop_loss_price'] ?? null;
            $tp1   = $levels['tp1_price'] ?? null;

            if ($entry === null || $sl === null || $tp1 === null) {
                $reasonCodes[] = 'WS_LEVELS_INCOMPLETE';
                $eligBlockCodes[] = 'WS_LEVELS_INCOMPLETE';
            } else {
                $R = (float)$entry - (float)$sl;
                if ($R <= 0.0) {
                    $reasonCodes[] = 'WS_R_INVALID_NONPOSITIVE';
                    $eligBlockCodes[] = 'WS_R_INVALID_NONPOSITIVE';
                } elseif ($R < (float)$tickSize) {
                    $reasonCodes[] = 'WS_R_INVALID_LT_TICK';
                    $eligBlockCodes[] = 'WS_R_INVALID_LT_TICK';
                } else {
                    $rrEst = ((float)$tp1 - (float)$entry) / $R;
                    if ($rrEst < 1.3) {
                        $reasonCodes[] = 'WS_RR_TOO_LOW';
                        $eligBlockCodes[] = 'WS_RR_TOO_LOW';
                    }
                }
            }

            // derived additions
            $derived['roc20'] = ($roc20 !== null) ? round((float)$roc20, 4) : null;
            $derived['rvol20'] = ($volRatio !== null && is_numeric($volRatio)) ? round((float)$volRatio, 4) : null;

            if (!empty($eligBlockCodes)) {
                // keep candidate but disable trading (watch-only)
                $scoreTotal = 0.0;
                $tradeDisabled = true;
                $eligBlockCodes = array_values(array_unique(array_merge($eligBlockCodes, ['GL_POLICY_DROP'])));
            } else {
                // 2.5 Scoring (0..1)
                $pattern = (isset($r['score_total']) && is_numeric($r['score_total'])) ? (float)$r['score_total'] : 0.0;
                if ($pattern > 1.0) $pattern = $pattern / 100.0;
                $sPattern = $this->clamp01($pattern);

                $closeVsMa20 = ($ma20 !== null && (float)$ma20 > 0) ? (($close / (float)$ma20) - 1.0) : null;
                $sTrend = ($closeVsMa20 !== null) ? $this->norm01((float)$closeVsMa20, -0.02, 0.05) : 0.0;

                $sMomentum = ($roc20 !== null) ? $this->norm01((float)$roc20, -0.03, 0.12) : 0.0;
                $sVolume = ($volRatio !== null && is_numeric($volRatio)) ? $this->norm01((float)$volRatio, 1.0, 3.0) : 0.0;

                $entryI = (float)($levels['entry_trigger_price'] ?? 0);
                $slI = (float)($levels['stop_loss_price'] ?? 0);
                $stopPct = ($entryI > 0 && $entryI > $slI) ? (($entryI - $slI) / $entryI) : null;

                $invStop = ($stopPct !== null) ? (1.0 - $this->norm01((float)$stopPct, 0.01, 0.08)) : 0.0;
                $invAtr  = ($atrPct !== null) ? (1.0 - $this->norm01((float)$atrPct, 0.01, 0.12)) : 0.0;
                $sRisk = $this->clamp01(0.5 * $invStop + 0.5 * $invAtr);

                $scoreTotal = $this->clamp01(
                    (0.30 * $sPattern) +
                    (0.25 * $sTrend) +
                    (0.20 * $sMomentum) +
                    (0.15 * $sVolume) +
                    (0.10 * $sRisk)
                );
            }
        } else {
            // policy hard filters and scoring (non-weekly)
            $policyRes = $this->applyPolicyRules(
                $policy,
                [
                    'ticker_id' => $tickerId,
                    'ticker_code' => $tickerCode,
                    'close' => $close,
                    'open' => $open,
                    'high' => $high,
                    'low' => $low,
                    'ma20' => $ma20,
                    'ma50' => $ma50,
                    'ma200' => $ma200,
                    'rsi14' => $rsi14,
                    'atr14' => $atr14,
                    'atr_pct' => $atrPct,
                    'tick_pct' => $tickPct,
                    'vol_ratio' => $volRatio,
                    'vol_sma20' => $r['vol_sma20'] ?? null,
                    'liq_bucket' => $liqBucket,
                    'dv20' => $dv20,
                    'signal_age_days' => $r['signal_age_days'] ?? null,
                    'setup_type' => $setupType,
                    'div_event' => $divEventsByTicker[$tickerId] ?? null,
                    'exec_trade_date' => $execTradeDate,
                    'candle' => $candle,
                    'prev_close' => $r['prev_close'] ?? null,
                ],
                $reasonCodes
            );

            if (($policyRes['drop'] ?? false) === true) {
                // Never DROP silently: keep candidate as WATCH_ONLY with score_total=0 for auditability.
                $reasonCodes = (array)($policyRes['reason_codes'] ?? $reasonCodes);
                $scoreTotal = 0.0;
                $entryStyle = (string)($policyRes['entry_style'] ?? 'Default');
                $eligBlockCodes = array_values(array_unique(array_merge((array)($policyRes['eligibility_block_codes'] ?? []), ['GL_POLICY_DROP'])));
                $tradeDisabled = true;
            } else {
                $reasonCodes = (array)($policyRes['reason_codes'] ?? $reasonCodes);
                $scoreTotal = (isset($r['score_total']) && is_numeric($r['score_total'])) ? (float)$r['score_total'] : 0.0;
                $entryStyle = (string)($policyRes['entry_style'] ?? 'Default');
                $eligBlockCodes = (array)($policyRes['eligibility_block_codes'] ?? []);
            }
